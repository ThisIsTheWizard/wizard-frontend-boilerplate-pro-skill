################################################################################
# snippet-template.txt
# Collapsible code-snippet block format for the /library showcase pages.
#
# Usage: copy the relevant FRAMEWORK BLOCK below into a category page.
# Replace every {{PLACEHOLDER}} with the real value for the component.
#
# Placeholders
#   {{COMPONENT_NAME}}    PascalCase component name, e.g. "Button"
#   {{COMPONENT_SLUG}}    kebab-case anchor id, e.g. "button", "radio-group"
#   {{COMPONENT_DESC}}    1-2 sentence plain-English description
#   {{LANGUAGE}}          tsx | vue | svelte
#   {{FILENAME}}          e.g. "Button.tsx", "Button.vue", "Button.svelte"
#   {{SNIPPET}}           Minimal self-contained usage example (5-15 lines).
#                         Must include imports. Must match the live preview.
################################################################################


################################################################################
# REACT / NEXT.JS  (TSX)
################################################################################

<section id="{{COMPONENT_SLUG}}">
  <h2>{{COMPONENT_NAME}}</h2>
  <p>{{COMPONENT_DESC}}</p>

  {/* Live preview */}
  <div className="rounded-lg border border-border bg-card p-8 flex items-center justify-center">
    {/* Render the live example here */}
  </div>

  {/* Collapsible code block */}
  <CodeBlock
    code={`{{SNIPPET}}`}
    collapsible
    defaultOpen={false}
    filename="{{FILENAME}}"
    language="{{LANGUAGE}}"
  />
</section>

# Minimal snippet shape (TSX):
#
#   import { {{COMPONENT_NAME}} } from "@/components/ui/{{COMPONENT_SLUG}}";
#
#   export default function {{COMPONENT_NAME}}Demo() {
#     return (
#       <{{COMPONENT_NAME}}>Label</{{COMPONENT_NAME}}>
#     );
#   }
#

################################################################################
# CODEBLOCK PROPS REFERENCE
#
# Prop            Type      Default   Notes
# --------------- --------- --------- -------------------------------------------
# code            string    required  Raw source text; shown verbatim in <pre>
# language        string    "tsx"     Label only (no runtime highlighting)
# filename        string    undefined Chip above the block (font-mono, xs, muted)
# collapsible     boolean   false     Wrap in expand/collapse toggle
# defaultOpen     boolean   true      Initial open state when collapsible=true
# highlightedCode string    undefined Pre-highlighted HTML rendered with raw
#                                     innerHTML; only use with trusted content
# className       string    undefined Extra classes on the outer wrapper
#
# Behaviour
#   - Collapsed label : "Show code" / expanded: "Hide code"
#   - aria-expanded on toggle button; aria-controls points to <pre> id
#   - Copy button     : top-right corner
#                       aria-label "Copy code" -> "Copied!" for 2 s on success
#   - <pre>           : role="region"
#                       aria-label="{filename} code snippet" (or "Code snippet")
#   - Zero external dependencies -- no syntax-highlighting library required
################################################################################
